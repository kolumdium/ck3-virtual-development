namespace = vdev_county

# This Event should be triggered for a person and a county
vdev_county.1 = {
    type = character_event
    # scope = landed_title
    hidden = no

    theme = education

    desc = vdev_county.1.desc

    # Set to Normal
    option = {
        name = vdev_change_county_focus_event.1.a
        trigger = { gold >= vdev_focus_change_gold_cost }
        remove_short_term_gold = vdev_focus_change_gold_cost
        scope:barony.county = {
            vdev_set_county_focus = { FOCUS = 0 }
        }
    }
    option = {
        name = vdev_change_county_focus_event.1.b
        trigger = { gold >= vdev_focus_change_gold_cost }
        remove_short_term_gold = vdev_focus_change_gold_cost
        scope:barony.county = {
            vdev_set_county_focus = { FOCUS = 1 }
        }
    }
    option = {
        name = vdev_change_county_focus_event.1.c
        trigger = { gold >= vdev_focus_change_gold_cost }
        remove_short_term_gold = vdev_focus_change_gold_cost
        scope:barony.county = {
            vdev_set_county_focus = { FOCUS = 2 }
        }
    }
    option = {
        name = vdev_change_county_focus_event.1.d
        trigger = { gold >= vdev_focus_change_gold_cost }
        remove_short_term_gold = vdev_focus_change_gold_cost
        scope:barony.county = {
            vdev_set_county_focus = { FOCUS = 3 }
        }
    }
    option = {
        name = vdev_change_county_focus_event.1.e
        trigger = { gold >= vdev_focus_change_gold_cost }
        remove_short_term_gold = vdev_focus_change_gold_cost
        scope:barony.county = {
            vdev_set_county_focus = { FOCUS = 4 }
        }
    }
    ## just close to not get stuck
    option = {
        name = vdev_change_county_focus_event.1.f
        is_cancel_option = yes
    }
}

vdev_county.3 = {
    type = character_event
    hidden = no
    theme = education
    desc = vdev_county.3.desc

    immediate = {
        scope:barony ?= {
            save_scope_as = vdev_current_barony
            root = {
                add_to_variable_list = {
                    name = vdev_viewed_baronies
                    target = scope:barony
                }
            }
            county = {
                save_scope_as = vdev_selected_county
            }
        }
        clear_saved_scope = vdev_next_barony
        ordered_held_title = {
            limit = {
                tier = tier_barony
                barony_is_valid_for_vdev_focus_trigger = {
                    CHARACTER = root
                }
                save_temporary_scope_as = vdev_candidate_barony
                root = {
                    NOT = {
                        is_target_in_variable_list = {
                            name = vdev_viewed_baronies
                            target = scope:vdev_candidate_barony
                        }
                    }
                }
            }
            order_by = development_level
            position = 0
            save_scope_as = vdev_next_barony
        }
    }

    option = {
        name = vdev_show_county_vdev_event.1.b
        trigger = {
            exists = scope:vdev_next_barony
        }
        scope:vdev_next_barony = {
            save_scope_as = barony
        }
        trigger_event = vdev_county.3
    }

    option = {
        name = vdev_show_county_vdev_event.1.a
        clear_variable_list = vdev_viewed_baronies
        is_cancel_option = yes
    }

}
