#root is the barony
barony_is_valid_for_vdev_convert_trigger = {
    holder = $CHARACTER$
    barony.county = {
        has_variable = vdev_county_normal
        var:vdev_county_normal >= 25
    }
}

# Root is the converting character; scope:barony is selected by the decision widget.
vdev_can_convert_selected_barony_trigger = {
    gold >= vdev_convert_vdev_gold_cost
    scope:barony ?= {
        barony_is_valid_for_vdev_convert_trigger = {
            CHARACTER = root
        }
    }
}

barony_is_valid_for_vdev_focus_trigger = {
    holder = $CHARACTER$
    county = {
        has_variable = vdev_county_focus
    }
}

barony_is_valid_for_vdev_building_trigger = {
    holder = $CHARACTER$
    title_province = {
        free_building_slots > 0
    }
    OR = {
        AND = {
            county = { var:vdev_county_normal >= 25 }
            title_province = {
                NOT = { has_building_or_higher = vdev_special_normal_01 }
            }
        }
        AND = {
            county = { var:vdev_county_offensive >= 25 }
            title_province = {
                NOT = { has_building_or_higher = vdev_special_offensive_01 }
            }
        }
        AND = {
            county = { var:vdev_county_defensive >= 25 }
            title_province = {
                NOT = { has_building_or_higher = vdev_special_defensive_01 }
            }
        }
        AND = {
            county = { var:vdev_county_health >= 25 }
            title_province = {
                NOT = { has_building_or_higher = vdev_special_health_01 }
            }
        }
        AND = {
            county = { var:vdev_county_infrastructure >= 25 }
            title_province = {
                NOT = { has_building_or_higher = vdev_special_infrastructure_01 }
            }
        }
    }
}

vdev_can_build_in_selected_barony_trigger = {
    gold >= vdev_building_gold_cost
    scope:barony ?= {
        barony_is_valid_for_vdev_building_trigger = {
            CHARACTER = root
        }
    }
}

barony_is_valid_for_vdev_building_slot_trigger = {
    holder = $CHARACTER$
    county = {
        has_variable = vdev_county_normal
        var:vdev_county_normal >= vdev_building_slot_vdev_cost
        var:vdev_extra_building_slots <= 5
    }
    NOT = {
        title_province = {
            building_slots > define:NProvince|MAX_BUILDINGS
        }
    }
}

vdev_can_add_slot_to_selected_barony_trigger = {
    gold >= vdev_building_slot_gold_cost
    scope:barony ?= {
        barony_is_valid_for_vdev_building_slot_trigger = {
            CHARACTER = root
        }
    }
}

barony_has_valid_vdev_amount_trigger = {
    holder = $CHARACTER$
    barony.county = {
        has_variable = $BUFF$
        var:$BUFF$ >= $AMOUNT$
    }
}
